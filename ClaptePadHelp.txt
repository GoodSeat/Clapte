# 「ヘルプの挿入」でいつでも参照できます。

# ------------------------------------------------------------------
# 1. 基本的な計算機能
# ------------------------------------------------------------------
# 左側に数式を入力します。自動で計算され、右側に答えが表示されます。
# 数式が複雑な場合には、結果の表示に時間がかかることがあります。
5cm * 10m

# 数式の最後に'='を入力すると、計算式を含んだ結果が表示されます。
5cm * 10m =

# '#'より後方はコメントとみなされ、評価されません。
# 5cm * 10m

# 長い数式などで、複数行にわたって数式を記述したい場合、
# 継続行の末尾に、スペースを空けて"_"を入力してください。
1 + 2 + 3 + 4 + 5 + _
6 + 7 + 8 + 9 + 10

# ------------------------------------------------------------------
# 2. 定数の定義
# ------------------------------------------------------------------
# 以下のように宣言して、定数を定義できます。
# (この定数は、計算機内の宣言の後方だけで有効です。)
a = 5m # テスト用定数

a^2

# 定数は何度でも再定義でき、数式ではより直前に宣言された定義が参照さ
# れます。
a = 10m

a^2 # ← a=10m が使用されます。


# 数式の前方に "|" を入力した行は「優先解決行」となります。
# 「優先解決行」は、直近の連続する非優先解決行よりも先に評価されます。
# 優先解決行を利用すると、定数の定義を、その定数を利用する行よりも後
# に置くといったことが可能です。
A  = B + C
A' = B / C
| # ここで、
| B = 7 cm^2
| C = D * E
| | D = 3 in   # 優先解決行をネストすることも可能です。
| | E = 20 mm

# ------------------------------------------------------------------
# 3. 関数の定義
# ------------------------------------------------------------------
# 以下のように宣言して、関数を定義できます。
# (この関数は、計算機内の宣言の後方だけで有効です。)
testFunction(a, b) = a * 5 + b # テスト用の関数

testFunction(1, 2)

# ------------------------------------------------------------------
# 4. 方程式の求解と定数の定義
# ------------------------------------------------------------------
# 方程式が入力された場合、その解が表示されます。
x^2 + 2*x + 1 = 0

# 方程式の求解は、同時に定数の定義とみなされます。
x * 5 # ←xには、上の方程式で求められた解が用いられます。


# 計算機では、連立方程式の求解も可能です。
# 連立方程式を解くには、先頭に "{" を入力して、対象とする等式を1行ず
# つ記述します。ただし、最後の数式の先頭には "{_" を記述します。
{  α + β = 8
{  β + γ = 4
{_ α^2   = 4

# 算出対象とする変数は自動で判定されますが、以下のようにして求解対象
# の変数を明示的に指定することも可能です。
{  x + z = 8
{  x + y = 4
{_ x, y


# ※定数の定義を行わず、単純に方程式の求解のみを行いたい場合には、
#   solve、solveN、solveB関数を利用してください。
solveN(sin(X) + cos(X) = 0.12345, X, 1.0E-10)

#   以下のように、求解結果を定数に入れておくと、"@"演算子を用いること
#   で求解結果を参照することも可能です。
ans = solveB(testFunction(X, 2) = 0.12345, X)
x0 = ans@1 # 解の参照
testFunction(x0, 2) = # 検算

# ------------------------------------------------------------------
# 5. 評価有無の制御
# ------------------------------------------------------------------
# $if、$elif、$else、$endif の各キーワードにより、条件に応じて
# 計算を実行するか否かを制御することができます。
# 例えば、以下の"cond"の値を変更すると、値に応じて"val"が変化します。
cond = 0 # 何らかの条件変数

$if cond = 0
  val = 3
$elif cond > 0
  val = 5
$else
  val = 7
$endif

val * 2 # cond = 0なら6、cond > 0なら10、cond < 0なら14

# ------------------------------------------------------------------
# 6. 各行の計算結果参照
# ------------------------------------------------------------------
# "_ans"あるいは"_ANS"変数により、前方の行の計算結果を参照できます。
# "_ans@n"にはn行目の計算結果が格納されています。
# "_ANS@n"にはn行上の計算結果が格納されています。
# また、"_LN"変数により、現在行番号を参照できます。
# 例えば、以下のように定義すると、"allSum"定数の値は"start"の定義行
# から"end"の定義行の間にある数式の総和となります。

start = _LN
100 m
200 m
300 m
400 m * 2
1 km
end = _LN

allSum = sigma(_ans@i, i=start+1, end-1)

# ------------------------------------------------------------------
# 7. 入力補助機能
# ------------------------------------------------------------------

# 例えば次の行に、"t"を入力すると、"tan"や"testFunction"等の候補が
# 表示されると思います。

# 入力補助は、説明でも引き当てることができます。
# 例えば次の行に、"対数"を入力すると、"log"や"ln"等の候補が、
# "テスト"を入力すれば、"a"や"testFunction"等の候補が表示されます。
